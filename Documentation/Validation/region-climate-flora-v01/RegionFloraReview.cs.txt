using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Collections.Generic;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;
using CityForgeV3.World;
using CityForgeV3.UI;
[InitializeOnLoad] public static class RegionFloraReview
{
 const string Root="/Users/joelinstrum/dev/CityForge-Regions-Review/";
 const BindingFlags Flags=BindingFlags.Instance|BindingFlags.NonPublic;
 static RegionFloraGeneration job; static RegionSaveData stress; static List<double> timings; static double total;
 static RegionFloraReview(){EditorApplication.update+=Poll;}
 static void Poll()
 {
  if(EditorApplication.isCompiling||EditorApplication.isUpdating)return;
  if(job!=null){StressStep();return;}
  var path=Root+"region-flora-review.request";if(!File.Exists(path))return;
  var command=File.ReadAllText(path).Trim();File.Delete(path);
  try
  {
   if(command=="tests")RunTests();
   if(command=="stress")
   {
    stress=RegionSaveStore.Create("Flora stress fixture",28,20);
    RegionRiverGenerator.Apply(stress,RegionRiverGenerator.Generate(stress,new RegionTerrainSettings{DeepRivers=RegionWaterAmount.Many,Streams=RegionWaterAmount.Many},159));
    job=new RegionFloraGeneration(stress,RegionTreeCoverage.Wooded,9527);timings=new();total=0;
   }
   if(command=="ui")Ui();
   if(command=="ui-generate")UiGenerate();
  }
  catch(Exception e){File.WriteAllText(Root+"region-flora-review-error.txt",e.ToString());}
 }
 static void RunTests()
 {
  var asm=Assembly.LoadFrom(Root+"Library/ScriptAssemblies/CityForgeV3.Tests.EditMode.dll");var lines=new List<string>();
  foreach(var name in new[]{"RegionFloraGeneratorTests","RegionTerrainMenuTests","RegionRiverGeneratorTests","RegionRiverNetworkTests","RegionDeletionTests"})
  {
   var type=asm.GetTypes().FirstOrDefault(t=>t.Name==name);if(type==null){lines.Add("FAIL missing fixture "+name);continue;}
   foreach(var method in type.GetMethods())
   {
    var attrs=method.GetCustomAttributes(false);var cases=attrs.Where(a=>a.GetType().Name=="TestCaseAttribute").ToArray();
    if(cases.Length==0&&!attrs.Any(a=>a.GetType().Name=="TestAttribute"))continue;
    var args=cases.Length==0?new[]{Array.Empty<object>()}:cases.Select(a=>(object[])a.GetType().GetProperty("Arguments").GetValue(a)).ToArray();
    foreach(var values in args)
    {
     var fixture=Activator.CreateInstance(type);
     try
     {
      foreach(var setup in type.GetMethods().Where(m=>m.GetCustomAttributes(false).Any(a=>a.GetType().Name=="SetUpAttribute")))setup.Invoke(fixture,null);
      method.Invoke(fixture,values);lines.Add("PASS "+name+"."+method.Name+" "+string.Join(",",values));
     }
     catch(Exception e){lines.Add("FAIL "+name+"."+method.Name+" "+(e.InnerException??e));}
     finally{foreach(var cleanup in type.GetMethods().Where(m=>m.GetCustomAttributes(false).Any(a=>a.GetType().Name=="TearDownAttribute")))cleanup.Invoke(fixture,null);}
    }
   }
  }
  File.WriteAllLines(Root+"region-flora-tests.txt",lines);
 }
 static void StressStep()
 {
  try
  {
   var watch=System.Diagnostics.Stopwatch.StartNew();job.Step();watch.Stop();timings.Add(watch.Elapsed.TotalMilliseconds);total+=watch.Elapsed.TotalMilliseconds;
   if(!job.Ready)return;
   job.Commit(_=>{});
   var path=Root+"ReviewScratch/FloraStress";
   watch.Restart();var save=RegionSaveStore.Save(stress,path);watch.Stop();var saveMs=watch.Elapsed.TotalMilliseconds;
   watch.Restart();var loaded=RegionSaveStore.Load(stress.RegionId,path);watch.Stop();var loadMs=watch.Elapsed.TotalMilliseconds;
   var count=loaded.Tiles.Sum(d=>d.Flora.Count);if(count!=job.TreeCount)throw new Exception("Stress reload count mismatch");
   var largest=loaded.Tiles.OrderByDescending(d=>d.Flora.Count).First();
   watch.Restart();var index=DistrictHarvestIndex.For(largest);var near=index.Nearby(Vector2.zero,100).Count();watch.Stop();
   var lines=new[]{"PASS full region generation and save/reload",$"Region: 28 x 20, districts: {stress.Tiles.Count}, trees: {job.TreeCount}",
    $"Generation CPU total: {total:0.00} ms; max district step: {timings.Max():0.00} ms; median step: {timings.OrderBy(v=>v).ElementAt(timings.Count/2):0.00} ms",
    $"Save: {saveMs:0.00} ms; reload: {loadMs:0.00} ms; bytes: {new FileInfo(save).Length}",
    $"Largest district: {largest.Width} x {largest.Height}, trees: {largest.Flora.Count}; harvest index build + 100m query: {watch.Elapsed.TotalMilliseconds:0.00} ms; candidates: {near}"};
   File.WriteAllLines(Root+"region-flora-stress.txt",lines);
   File.Delete(save);job=null;stress=null;
  }
  catch(Exception e){job=null;stress=null;File.WriteAllText(Root+"region-flora-review-error.txt",e.ToString());}
 }
 static void Submit(Button button)
 {
  if(button==null)throw new Exception("Missing button");
  using(var e=NavigationSubmitEvent.GetPooled()){e.target=button;button.SendEvent(e);}
 }
 static void UiGenerate()
 {
  var app=UnityEngine.Object.FindFirstObjectByType<CityForgeApp>();var type=app.GetType();
  var root=(VisualElement)type.GetField("_root",Flags).GetValue(app);
  var old=(RegionSaveData)type.GetField("_openRegion",Flags).GetValue(app);
  var region=new RegionSaveData{RegionId="flora-ui-generation-"+Guid.NewGuid().ToString("N"),Name="Flora generation fixture",Width=8,Height=8,
   Tiles=new(){new RegionCityTile{TileId="forest",Name="Forest",Width=2,Height=2}}};
  var show=type.GetMethod("Show",Flags);var screen=Enum.Parse(show.GetParameters()[0].ParameterType,"RegionEditor");
  Action modal=()=>type.GetMethod("ComposeRegionTerrainModal",Flags).Invoke(app,null);
  type.GetField("_openRegion",Flags).SetValue(app,region);show.Invoke(app,new[]{screen});modal();
  Submit(root.Q<Button>("region-terrain-flora"));Submit(root.Q<Button>("region-flora-wooded"));Submit(root.Q<Button>("generate-region-flora"));
  int attempts=0;
  void Check()
  {
   if(root.Q<VisualElement>("region-terrain-modal")!=null&&attempts++<100){root.schedule.Execute(Check).ExecuteLater(100);return;}
   try
   {
    if(region.Tiles[0].Flora.Count<1500)throw new Exception("Scheduled generation did not finish: "+region.Tiles[0].Flora.Count);
    var loaded=RegionSaveStore.Load(region.RegionId);if(loaded==null||loaded.Tiles[0].Flora.Count!=region.Tiles[0].Flora.Count)throw new Exception("UI-generated trees did not persist");
    File.AppendAllText(Root+"region-flora-ui.txt",$"PASS scheduled Wooded generation, auto-save and reload: {loaded.Tiles[0].Flora.Count} trees\n");
   }
   catch(Exception e){File.WriteAllText(Root+"region-flora-review-error.txt",e.ToString());}
   finally
   {
    if(File.Exists(Path.Combine(RegionSaveStore.DefaultRoot,region.RegionId+".json")))RegionSaveStore.Delete(region.RegionId);
    type.GetField("_openRegion",Flags).SetValue(app,old);show.Invoke(app,new[]{screen});modal();Submit(root.Q<Button>("region-terrain-flora"));
    root.schedule.Execute(()=>ScreenCapture.CaptureScreenshot(Root+"region-flora-menu.png")).ExecuteLater(500);
   }
  }
  root.schedule.Execute(Check).ExecuteLater(100);
 }
 static void Ui()
 {
  var app=UnityEngine.Object.FindFirstObjectByType<CityForgeApp>();var type=app.GetType();
  var root=(VisualElement)type.GetField("_root",Flags).GetValue(app);
  if(root==null)throw new Exception("Restart isolated Play mode before UI checks");
  var old=(RegionSaveData)type.GetField("_openRegion",Flags).GetValue(app);
  var region=RegionSaveStore.Create("Flora UI fixture",8,8);region.Terrain.Climate=RegionClimate.Desert;
  var oldTile=type.GetField("_selectedRegionTileId",Flags).GetValue(app);
  type.GetField("_openRegion",Flags).SetValue(app,region);
  var show=type.GetMethod("Show",Flags);var regionScreen=Enum.Parse(show.GetParameters()[0].ParameterType,"RegionEditor");
  Action modal=()=>type.GetMethod("ComposeRegionTerrainModal",Flags).Invoke(app,null);
  try
  {
   show.Invoke(app,new[]{regionScreen});modal();
   Submit(root.Q<Button>("region-terrain-flora"));
   if(root.Q<Button>("region-flora-sparse").enabledSelf||root.Q<Button>("region-flora-wooded").enabledSelf||root.Q<Button>("generate-region-flora").enabledSelf)throw new Exception("Desert coverage not disabled");
   Submit(root.Q<Button>("region-terrain-climate"));Submit(root.Q<Button>("region-climate-mediterranean"));Submit(root.Q<Button>("save-region-climate"));
   if(region.Terrain.Climate!=RegionClimate.Mediterranean)throw new Exception("Climate not applied");
   var saved=RegionSaveStore.Load(region.RegionId);if(saved.Terrain.Climate!=RegionClimate.Mediterranean)throw new Exception("Climate not persisted");
   modal();Submit(root.Q<Button>("region-terrain-flora"));Submit(root.Q<Button>("region-flora-wooded"));
   if(!root.Q<Button>("generate-region-flora").enabledSelf)throw new Exception("Wooded not enabled");
   // Generation is intentionally not committed to the user's region; this fixture is disposable.
   Submit(root.Q<Button>("generate-region-flora"));Submit(root.Q<Button>("cancel-region-terrain-options"));
   if(region.Tiles.Any(d=>d.Flora.Count!=0))throw new Exception("Cancel mutated region");
   File.WriteAllText(Root+"region-flora-ui.txt","PASS Desert disables Sparse, Wooded and Generate\nPASS Mediterranean climate button saves/reloads\nPASS Wooded enables generation\nPASS cancel leaves fixture trees untouched\n");
  }
  finally
  {
   if(File.Exists(Path.Combine(RegionSaveStore.DefaultRoot,region.RegionId+".json")))RegionSaveStore.Delete(region.RegionId);
   type.GetField("_openRegion",Flags).SetValue(app,old);type.GetField("_selectedRegionTileId",Flags).SetValue(app,oldTile);
   show.Invoke(app,new[]{regionScreen});modal();Submit(root.Q<Button>("region-terrain-climate"));
   root.schedule.Execute(()=>ScreenCapture.CaptureScreenshot(Root+"region-climate-menu.png")).ExecuteLater(500);
  }
 }
}
