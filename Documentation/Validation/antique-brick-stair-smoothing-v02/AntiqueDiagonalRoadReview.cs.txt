using System.IO;
using System.Collections.Generic;
using CityForgeV3.World;
using UnityEditor;
using UnityEngine;

public static class AntiqueDiagonalRoadReview
{
    public static void Profile()
    {
        var rows = new List<PlacedRoadPiece>();
        for (var z = 8; z < 56; z++)
            for (var x = 8; x < 56; x++)
                rows.Add(new PlacedRoadPiece
                {
                    GridX = x, GridZ = z,
                    PackageId = RoadPiecePackageCatalog.TwoLaneSidewalkId,
                    RoadMaterialId = "antique-brick",
                    SidewalkMaterialId = "antique-brick"
                });
        DistrictRoadPlacementModel.Repair(rows);
        var indexed = new List<PlacedRoadPiece>(rows);
        var scanned = new List<PlacedRoadPiece>(rows);
        var watch = new System.Diagnostics.Stopwatch();
        var allocation = System.GC.GetAllocatedBytesForCurrentThread();
        var session = new DistrictRoadPlacementModel.EditSession(indexed);
        var treasury = 100000;
        watch.Start();
        for (var x = 0; x < 30; x++)
            session.TryPlace(x, 60, 64, 64,
                DistrictRoadPlacementModel.AntiqueBrickFamily, ref treasury);
        watch.Stop();
        var localMs = watch.Elapsed.TotalMilliseconds;
        var localBytes = System.GC.GetAllocatedBytesForCurrentThread() - allocation;
        treasury = 100000;
        allocation = System.GC.GetAllocatedBytesForCurrentThread();
        watch.Restart();
        for (var x = 0; x < 30; x++)
            DistrictRoadPlacementModel.TryPlace(scanned, x, 60, 64, 64,
                DistrictRoadPlacementModel.AntiqueBrickFamily, ref treasury);
        watch.Stop();
        var rebuildMs = watch.Elapsed.TotalMilliseconds;
        var rebuildBytes = System.GC.GetAllocatedBytesForCurrentThread() - allocation;

        var district = new RegionCityTile { Width = 1, Height = 1, Roads = rows };
        var root = new GameObject("Dense Antique Brick Review");
        var target = new RenderTexture(800, 600, 24);
        try
        {
            var world = root.AddComponent<DistrictWorldController>();
            var buildWatch = System.Diagnostics.Stopwatch.StartNew();
            world.Build(district);
            buildWatch.Stop();
            var roadSession = new DistrictRoadPlacementModel.EditSession(rows);
            for (var index = 15; index < 45; index++)
                roadSession.TryConnectDiagonal(new Vector2Int(index, index),
                    new Vector2Int(index + 1, index + 1));
            world.RefreshRoads(district);
            var camera = world.WorldCamera;
            camera.orthographic = true;
            camera.orthographicSize = 200f;
            camera.transform.SetPositionAndRotation(
                new Vector3(0f, 800f, 0f), Quaternion.Euler(90f, 0f, 0f));
            camera.targetTexture = target;
            var samples = new List<double>();
            var beforeRenderBytes = System.GC.GetAllocatedBytesForCurrentThread();
            for (var index = 0; index < 20; index++)
            {
                watch.Restart();
                camera.Render();
                watch.Stop();
                samples.Add(watch.Elapsed.TotalMilliseconds);
            }
            samples.Sort();
            var renderBytes = System.GC.GetAllocatedBytesForCurrentThread() - beforeRenderBytes;
            File.WriteAllText("/tmp/cityforge-antique-diagonal-profile.txt",
                $"roads={rows.Count} diagonalLinks=30 placed=30\n" +
                $"local edit ms={localMs:F3} allocated={localBytes}\n" +
                $"per-edit full index proxy ms={rebuildMs:F3} allocated={rebuildBytes}\n" +
                $"build ms={buildWatch.Elapsed.TotalMilliseconds:F3}\n" +
                $"render 800x600 frames=20 medianMs={samples[10]:F3} p95Ms={samples[19]:F3} maxMs={samples[19]:F3} allocated={renderBytes} drawCalls={UnityStats.drawCalls} batches={UnityStats.batches}\n");
        }
        finally
        {
            Object.DestroyImmediate(target);
            Object.DestroyImmediate(root);
        }
    }

    public static void Capture()
    {
        var district = new RegionCityTile
        {
            Width = 1, Height = 1, TimeOfDay = TimeOfDayPreset.Noon
        };
        var session = new DistrictRoadPlacementModel.EditSession(district.Roads);
        var treasury = 1000;
        var path = new[]
        {
            new Vector2Int(27, 26), new Vector2Int(28, 26),
            new Vector2Int(29, 26), new Vector2Int(30, 26),
            new Vector2Int(31, 26), new Vector2Int(32, 26),
            new Vector2Int(33, 26), new Vector2Int(34, 26),
            new Vector2Int(34, 27), new Vector2Int(35, 27),
            new Vector2Int(35, 28), new Vector2Int(36, 28),
            new Vector2Int(36, 29), new Vector2Int(37, 29),
            new Vector2Int(37, 30), new Vector2Int(38, 30),
            new Vector2Int(38, 31), new Vector2Int(39, 31),
            new Vector2Int(39, 32), new Vector2Int(40, 32),
            new Vector2Int(40, 33)
        };
        var stroke = new List<Vector2Int>();
        var added = new HashSet<Vector2Int>();
        foreach (var cell in path)
        {
            session.TryPlace(cell.x, cell.y, 64, 64,
                DistrictRoadPlacementModel.AntiqueBrickFamily, ref treasury);
            stroke.Add(cell);
            added.Add(cell);
            session.TrySmoothAntiqueBrickStaircase(stroke, added,
                ref treasury, (_, _) => true, out _);
        }

        var root = new GameObject("Antique Diagonal Review");
        var target = new RenderTexture(1200, 900, 24);
        var image = new Texture2D(1200, 900, TextureFormat.RGB24, false);
        try
        {
            var world = root.AddComponent<DistrictWorldController>();
            world.Build(district);
            var camera = world.WorldCamera;
            camera.orthographic = true;
            camera.orthographicSize = 95f;
            camera.transform.SetPositionAndRotation(
                new Vector3(18f, 100f, -25f), Quaternion.Euler(90f, 0f, 0f));
            camera.targetTexture = target;
            camera.Render();
            RenderTexture.active = target;
            image.ReadPixels(new Rect(0, 0, 1200, 900), 0, 0);
            image.Apply();
            File.WriteAllBytes("/tmp/cityforge-antique-stair-road-review.png",
                image.EncodeToPNG());
        }
        finally
        {
            RenderTexture.active = null;
            Object.DestroyImmediate(image);
            Object.DestroyImmediate(target);
            Object.DestroyImmediate(root);
        }
    }
}
